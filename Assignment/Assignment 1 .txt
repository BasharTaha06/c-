class BankAccount
{
    private string _accountNumber;
    private decimal _balance;
    public BankAccount(string _accountNumber, decimal _balance)
    {
        this._accountNumber = _accountNumber;
        this._balance = _balance;
    }

    public string AccountNumber
    {
        get { return _accountNumber; }
    }
    public decimal Balance
    {
        get { return _balance; }
    }
    public void Deposit(decimal amount)
    {
        if (amount > 0)
        {
            _balance += amount;
        }
        else 
        { 
            Console.WriteLine("Deposit amount must be positive.");
        }
    }

    public void Withdraw(decimal amount)
    {
        if (amount > 0 && amount <= _balance)
        {
            _balance -= amount;
        }
        else
        {
            Console.WriteLine("Insufficient funds or invalid withdrawal amount.");
        }
    }

}


class Book
{
    private string title;
    private string isbn;
    private bool isCheckedOut=false;
    private string borrowerName="";

    public Book(string title, string isbn)
    {
        this.title = title;
        this.isbn = isbn;
    }

    public string Title
    {
        get { return title; }
    }

    public string ISBN
    {
        get { return isbn; }
    }
    public bool IsCheckedOut
    {
        get { return isCheckedOut; }
    }

    public string BorrowerName()
    {
        if(isCheckedOut)
        {
            return borrowerName;
        }
        else
        {
            return "Book is not checked out.";
        }
    }

    public void CheckOut(string borrowerName)
    {
        if (!isCheckedOut)
        {
            isCheckedOut = true;
            this.borrowerName = borrowerName;
        }
        else
        {
            Console.WriteLine("Book is already checked out.");
        }
    }

    public void ReturnBook()
    {
        if (isCheckedOut)
        {
            isCheckedOut = false;
            borrowerName = "";
        }
        else
        {
            Console.WriteLine("Book is not checked out.");
        }
    }

    public void GetStatus()
    {
        if(isCheckedOut)
        {
            Console.WriteLine($"Book '{title}' is checked out by {borrowerName}.");
        }
        else
        {
            Console.WriteLine($"Book '{title}' is available.");
        }
    }

}


class Account
{
    private string id;
    private string name;
    private int balance = 0;

    public Account(string id, string name)
    {
        this.id = id;
        this.name = name;
    }

    public Account(string id, string name, int balance)
    {
        this.id = id;
        this.name = name;
        this.balance = balance;
    }

    public string getId()
    {
        return id;
    }

    public string getName()
    {
        return name;
    }

    public int getBalance()
    {
        return balance;
    }

    public int credit(int amount)
    {
        balance += amount;
        return balance;
    }

    public int debit(int amount)
    {
        if (amount <= balance)
        {
            balance -= amount;

        }
        else
        {
            Console.WriteLine("Amount exceeded balance");
        }
        return balance;
    }

    public int transferTo(Account another, int amount)
    {
        if (amount <= balance)
        {
            balance -= amount;
            another.credit(amount);
        }
        else
        {
            Console.WriteLine("Amount exceeded balance");
        }
        return balance;
    }




    public string toString()
    {
        return $"Account[id={id}, name={name}, balance={balance}]";
    }
}
